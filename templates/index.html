function renderSidebar() { // Obtener el idioma activo en mayúsculas ('EN' o 'DE') const currentLang = document.getElementById('idiomaSelect').value.toUpperCase(); // 1. RENDERIZAR SESIONES (se muestran todas o puedes filtrarlas si lo prefieres) const sessionList = document.getElementById('sessionList'); if (sessionData.length === 0) { sessionList.innerHTML = '

No hay sesiones guardadas.

'; } else { sessionList.innerHTML = sessionData.map(s => `
📌 ${s.key.replace(/_/g, ' ')}
${s.fecha} | ${s.messages.length} mensajes
`).join(''); } // 2. RENDERIZAR VOCABULARIO (Filtrado solo por el idioma activo) const vocabList = document.getElementById('vocabList'); const filteredVocab = vocabData.filter(item => item.lang === currentLang); if (filteredVocab.length === 0) { vocabList.innerHTML = `

Sin vocabulario registrado para ${currentLang === 'EN' ? 'Inglés' : 'Alemán'}.

`; } else { const groupedVocab = {}; filteredVocab.forEach(item => { const ctx = item.context || 'General'; if (!groupedVocab[ctx]) groupedVocab[ctx] = []; groupedVocab[ctx].push(item); }); let htmlV = ''; for (const [ctx, items] of Object.entries(groupedVocab)) { htmlV += `
📁 Contexto: ${ctx} (${items.length})
`; items.forEach(v => { htmlV += `
💡 ${v.term}
`; }); htmlV += `
`; } vocabList.innerHTML = htmlV; } // 3. RENDERIZAR GRAMÁTICA (Filtrada solo por el idioma activo) const grammarList = document.getElementById('grammarList'); const filteredGrammar = grammarData.filter(item => item.lang === currentLang); if (filteredGrammar.length === 0) { grammarList.innerHTML = `

Sin gramática registrada para ${currentLang === 'EN' ? 'Inglés' : 'Alemán'}.

`; } else { const groupedGrammar = {}; filteredGrammar.forEach(item => { const top = item.topic || 'Estrategia/Regla'; if (!groupedGrammar[top]) groupedGrammar[top] = []; groupedGrammar[top].push(item); }); let htmlG = ''; for (const [top, items] of Object.entries(groupedGrammar)) { htmlG += `
📐 Tema: ${top} (${items.length})
`; items.forEach(g => { htmlG += `
🔹 ${g.rule}
`; }); htmlG += `
`; } grammarList.innerHTML = htmlG; } }
